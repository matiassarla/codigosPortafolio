Duplicar juicios de un período en otro

        
document.querySelectorAll('tr.filaalumno').forEach(tr => {
  // Buscar el juicio del Módulo Introductorio
  const tdIntro = [...tr.querySelectorAll('td')].find(td =>
    td.getAttribute('data-mescalif') === "Módulo
Introductorio"
  );

  // Buscar el textarea del módulo introductorio
  const introTextarea = tdIntro?.querySelector('textarea[name^="formHacerPromedios"]');
  const juicio = introTextarea?.value?.trim();

  // Buscar el td correspondiente a Abril
  const tdAbr = [...tr.querySelectorAll('td')].find(td =>
    td.getAttribute('data-mescalif') === "Abr"
  );

  // Buscar el textarea de abril y pegar el juicio
  const abrTextarea = tdAbr?.querySelector('textarea[name^="formHacerPromedios"]');
  if (juicio && abrTextarea) {
    abrTextarea.value = juicio;

    // Disparar eventos para que el sistema del sitio detecte el cambio
    abrTextarea.dispatchEvent(new Event('input', { bubbles: true }));
    abrTextarea.dispatchEvent(new Event('change', { bubbles: true }));
  }
});
