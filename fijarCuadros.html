Fijar cuadros de texto de juicio a un tamaño específico

        


function ajustarTextareas() {
  document.querySelectorAll('textarea').forEach(t => {
    const estilo = window.getComputedStyle(t);
    const visible = estilo.display !== "none" && estilo.visibility !== "hidden" && t.offsetParent !== null;
    if (visible) {
      t.style.width = "150px"; // Ajusta el ancho a 150px
      t.style.height = "150px"; // Ajusta la altura a 150px
    }
  });
}

// Ajustar al inicio
ajustarTextareas();

// Observar cambios en el DOM y volver a ajustar
const observer = new MutationObserver(() => {
  ajustarTextareas();
});

observer.observe(document.body, {
  childList: true,
  subtree: true
});
