Acredita o No acredita según calificación

        


let contador = 0;

document.querySelectorAll("tr.filaalumno").forEach(fila => {

  // Buscar el juicio de APE Febrero (puede ser otro período, ajustar según necesidad)
  const td = fila.querySelector(
    'td[data-mescalif="APE Febrero - Reunión Ficta EMB"]'
  );
  if (!td) return;

  const notaDiv = td.querySelector(".evalcon");
  const juicioDiv = td.querySelector(".evaljuicio");

  if (!notaDiv || !juicioDiv) return;

  // Intentar obtener nota desde input o texto
  let nota = null;

  const inputNota = notaDiv.querySelector("input");
  if (inputNota) {
    nota = parseFloat(inputNota.value.replace(",", "."));
  } else {
    nota = parseFloat(notaDiv.textContent.trim().replace(",", "."));
  }

  if (isNaN(nota)) return;

  const texto = nota >= 5 ? "Acredita." : "No acredita.";

  // Intentar escribir en juicio como si el usuario escribiera
  juicioDiv.focus();

  const inputJuicio = juicioDiv.querySelector("input, textarea");

  if (inputJuicio) {
    inputJuicio.value = texto;
    inputJuicio.dispatchEvent(new Event("input", { bubbles: true }));
    inputJuicio.dispatchEvent(new Event("change", { bubbles: true }));
  } else {
    juicioDiv.textContent = texto;
    juicioDiv.dispatchEvent(new Event("input", { bubbles: true }));
    juicioDiv.dispatchEvent(new Event("blur", { bubbles: true }));
  }

  contador++;
});

console.log("Procesados:", contador);
