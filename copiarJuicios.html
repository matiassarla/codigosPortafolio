Copiar juicios de un período para pegar en GSheets

        


(function () {
  const juicios = [];
  // Buscar el juicio del Módulo Introductorio (puede ser otro período, ajustar según necesidad)
  document.querySelectorAll('tr.filaalumno').forEach(tr => {
    const tdIntro = [...tr.querySelectorAll('td')].find(td =>
      td.getAttribute('data-mescalif') === "Módulo
Introductorio"
    );

    const introTextarea = tdIntro?.querySelector('textarea[name^="formHacerPromedios"]');
    const juicio = introTextarea?.value?.trim();

    juicios.push(juicio || "");
  });

  const textoFinal = juicios.join("\n");

  // Crear textarea temporal
  const textareaTemp = document.createElement("textarea");
  textareaTemp.value = textoFinal;
  document.body.appendChild(textareaTemp);

  textareaTemp.select();
  document.execCommand("copy");

  document.body.removeChild(textareaTemp);

  alert("Juicios copiados al portapapeles ✅");
})();
