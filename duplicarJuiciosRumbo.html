Duplicar juicios y notas de Módulo 1 en Módulo 3

        



let juiciosCopiados = 0;
let calificacionesCopiadas = 0;

document.querySelectorAll('tr.filaalumno').forEach(tr => {

  // Origen: Módulo Introductorio
  const tdIntro = tr.querySelector('td[data-mescalif="Módulo 1"]');
  if (!tdIntro) return;

  // Destino: Abril
  const tdAbr = tr.querySelector('td[data-mescalif="Módulo 3"]');
  if (!tdAbr) return;

  // =========================
  // COPIAR JUICIO
  // =========================

  const introTextarea = tdIntro.querySelector('textarea[name^="formHacerPromedios"]');
  const abrTextarea = tdAbr.querySelector('textarea[name^="formHacerPromedios"]');

  if (introTextarea && abrTextarea) {
    if (abrTextarea.value !== introTextarea.value) {
      abrTextarea.value = introTextarea.value;

      abrTextarea.dispatchEvent(new Event('input', { bubbles: true }));
      abrTextarea.dispatchEvent(new Event('change', { bubbles: true }));

      juiciosCopiados++;
    }
  }

  // =========================
  // COPIAR CALIFICACIÓN
  // =========================

  const notaOrigen = tdIntro.querySelector('.evalcon input');
  const notaDestino = tdAbr.querySelector('.evalcon input');

  if (notaOrigen && notaDestino) {
    if (notaDestino.value !== notaOrigen.value) {
      notaDestino.value = notaOrigen.value;

      notaDestino.dispatchEvent(new Event('input', { bubbles: true }));
      notaDestino.dispatchEvent(new Event('change', { bubbles: true }));

      calificacionesCopiadas++;
    }
  }

});

console.log(`Juicios copiados: ${juiciosCopiados}`);
console.log(`Calificaciones copiadas: ${calificacionesCopiadas}`);
console.log(`Total de cambios: ${juiciosCopiados + calificacionesCopiadas}`);
